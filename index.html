# Engineering data → interpretable result
def analyze_signal(time, amplitude, thickness):
    filtered = smooth(amplitude)
    peaks = detect_peaks(filtered)

    t1 = select_peak(peaks, window=1)
    t2 = select_peak(peaks, window=2)
    tof = t2 - t1

    velocity = (2 * thickness) / tof
    validate(tof, velocity)

    return {
      "TOF": tof,
      "velocity": velocity,
      "peaks": [t1, t2]
    }
